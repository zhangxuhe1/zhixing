```ts
import { SignJWT, jwtVerify } from "jose";
import { config } from "./config.js";

const secret = () => new TextEncoder().encode(config.sessionSecret);

export async function signSession(userId: string): Promise<string> {
  return new SignJWT({})
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(userId)
    .setIssuedAt()
    .setExpirationTime("180d")
    .sign(secret());
}

export async function verifySession(token: string): Promise<string> {
  const { payload } = await jwtVerify(token, secret());
  if (!payload.sub) throw new Error("session token missing sub");
  return payload.sub;
}
```

```ts
import { createRemoteJWKSet, jwtVerify, type JWTVerifyGetKey } from "jose";

export type AppleJwks = JWTVerifyGetKey;

export function remoteAppleJwks(): AppleJwks {
  return createRemoteJWKSet(new URL("https://appleid.apple.com/auth/keys"));
}

export async function verifyAppleIdentityToken(
  token: string,
  jwks: AppleJwks,
  bundleId: string,
): Promise<{ sub: string }> {
  const { payload } = await jwtVerify(token, jwks, {
    issuer: "https://appleid.apple.com",
    audience: bundleId,
  });
  if (!payload.sub) throw new Error("apple token missing sub");
  return { sub: payload.sub };
}
```

`apps/api/src/routes/auth.ts`:
```ts
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { verifyAppleIdentityToken } from "../apple.js";
import { requireUser } from "../auth.js";
import { config } from "../config.js";
import { signSession } from "../session.js";

const appleSchema = z.object({ identityToken: z.string().min(10), nickname: z.string().min(1).max(20).optional() });
const devSchema = z.object({ appleSub: z.string().min(1), nickname: z.string().min(1).max(20).optional() });

async function loginBySub(app: FastifyInstance, appleSub: string, nickname?: string) {
  const user = await app.prisma.user.upsert({
    where: { appleSub },
    update: nickname ? { nickname } : {},
    create: { appleSub, nickname: nickname ?? `旅人${appleSub.slice(-4)}` },
  });
  return { token: await signSession(user.id), user: { id: user.id, nickname: user.nickname } };
}

export async function authRoutes(app: FastifyInstance) {
  app.post("/auth/apple", async (req, reply) => {
    const body = appleSchema.safeParse(req.body);
    if (!body.success) return reply.code(400).send({ error: "invalid_body" });
    try {
      const { sub } = await verifyAppleIdentityToken(
        body.data.identityToken, app.appleJwks, config.appleBundleId,
      );
      return await loginBySub(app, sub, body.data.nickname);
    } catch {
      return reply.code(401).send({ error: "invalid_apple_token" });
    }
  });

  if (app.devAuth) {
    app.post("/auth/dev", async (req, reply) => {
      const body = devSchema.safeParse(req.body);
      if (!body.success) return reply.code(400).send({ error: "invalid_body" });
      return loginBySub(app, body.data.appleSub, body.data.nickname);
    });
  }

  app.get("/me", { preHandler: requireUser }, async (req, reply) => {
    const user = await app.prisma.user.findUnique({ where: { id: req.userId } });
    if (!user) return reply.code(401).send({ error: "unauthorized" });
    return { id: user.id, nickname: user.nickname };
  });
}
```

`apps/api/src/storage.ts`:
```ts
import {
  CreateBucketCommand, DeleteObjectCommand, GetObjectCommand,
  HeadBucketCommand, PutObjectCommand, S3Client,
} from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";

export interface StorageConfig {
  endpoint: string;
  publicEndpoint: string; // 客户端可达的地址，presign 用它
  region: string;
  bucket: string;
  accessKey: string;
  secretKey: string;
}

export type Storage = ReturnType<typeof createStorage>;

export function createStorage(cfg: StorageConfig) {
  const mkClient = (endpoint: string) =>
    new S3Client({
      endpoint,
      region: cfg.region,
      forcePathStyle: true,
      credentials: { accessKeyId: cfg.accessKey, secretAccessKey: cfg.secretKey },
    });
  const internal = mkClient(cfg.endpoint);
  const publik = mkClient(cfg.publicEndpoint);

  return {
    async ensureBucket() {
      try {
        await internal.send(new HeadBucketCommand({ Bucket: cfg.bucket }));
      } catch {
        await internal.send(new CreateBucketCommand({ Bucket: cfg.bucket }));
      }
    },
    async put(key: string, body: Buffer, contentType: string) {
      await internal.send(
        new PutObjectCommand({ Bucket: cfg.bucket, Key: key, Body: body, ContentType: contentType }),
      );
    },
    async presignGet(key: string, expiresSec = 300) {
      return getSignedUrl(publik, new GetObjectCommand({ Bucket: cfg.bucket, Key: key }), {
        expiresIn: expiresSec,
      });
    },
    async remove(keys: string[]) {
      await Promise.all(
        keys.map((key) => internal.send(new DeleteObjectCommand({ Bucket: cfg.bucket, Key: key }))),
      );
    },
  };
}
```

```ts
import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { requireUser } from "../auth.js";

const createSchema = z.object({
  placeId: z.string(),
  text: z.string().max(2000).optional(),
  date: z.string().datetime({ offset: true }),
});

const includeShape = {
  place: { select: { id: true, name: true } },
  photos: { select: { id: true, takenAt: true } },
} as const;

type PlaceNode = { id: string; parentId: string | null; level: string };

export function rollup(placeIds: string[], byId: Map<string, PlaceNode>) {
  const cities = new Set<string>();
  const provinces = new Set<string>();
  for (const placeId of placeIds) {
    let city: string | null = null;
    let province: string | null = null;
    let top = placeId;
    let cur = byId.get(placeId);
    while (cur) {
      if (cur.level === "CITY") city = cur.id;
      if (cur.level === "PROVINCE") province = cur.id;
      top = cur.id;
      cur = cur.parentId ? byId.get(cur.parentId) : undefined;
    }
    cities.add(city ?? top);
    if (province) provinces.add(province);
  }
  return { cities: cities.size, provinces: provinces.size };
}

export async function footprintsRoutes(app: FastifyInstance) {
  app.post("/footprints", { preHandler: requireUser }, async (req, reply) => {
    const body = createSchema.safeParse(req.body);
    if (!body.success) return reply.code(400).send({ error: "invalid_body", detail: body.error.flatten() });
    const place = await app.prisma.place.findUnique({ where: { id: body.data.placeId } });
    if (!place) return reply.code(400).send({ error: "unknown_place" });
    const footprint = await app.prisma.footprint.create({
      data: {
        userId: req.userId,
        placeId: body.data.placeId,
        text: body.data.text,
        date: new Date(body.data.date),
      },
      include: includeShape,
    });
    return reply.code(201).send(footprint);
  });

  app.get("/me/footprints", { preHandler: requireUser }, async (req) => {
    return app.prisma.footprint.findMany({
      where: { userId: req.userId },
      orderBy: { date: "desc" },
      include: includeShape,
    });
  });

  app.delete("/footprints/:id", { preHandler: requireUser }, async (req, reply) => {
    const { id } = req.params as { id: string };
    const footprint = await app.prisma.footprint.findUnique({
      where: { id }, include: { photos: { select: { mediaUrl: true } } },
    });
    if (!footprint || footprint.userId !== req.userId) {
      return reply.code(404).send({ error: "not_found" });
    }
    await app.prisma.footprint.delete({ where: { id } }); // photos 级联删除
    app.storage.remove(footprint.photos.map((p) => p.mediaUrl)).catch(() => {
      app.log.warn({ id }, "storage cleanup failed");
    });
    return reply.code(204).send();
  });

  app.get("/me/stats", { preHandler: requireUser }, async (req) => {
    const [footprints, fpPlaces, places] = await Promise.all([
      app.prisma.footprint.count({ where: { userId: req.userId } }),
      app.prisma.footprint.findMany({
        where: { userId: req.userId }, select: { placeId: true }, distinct: ["placeId"],
      }),
      app.prisma.place.findMany({ select: { id: true, parentId: true, level: true } }),
    ]);
    const byId = new Map(places.map((p) => [p.id, p]));
    const placeIds = fpPlaces.map((f) => f.placeId);
    return { footprints, places: placeIds.length, ...rollup(placeIds, byId) };
  });
}
```

```ts
import { randomUUID } from "node:crypto";
import type { FastifyInstance } from "fastify";
import type { MultipartFields } from "@fastify/multipart";
import { requireUser } from "../auth.js";

const EXT_BY_MIME: Record<string, string> = {
  "image/jpeg": "jpg",
  "image/png": "png",
  "image/heic": "heic",
};

function fieldValue(fields: MultipartFields, name: string): string | undefined {
  const f = fields[name];
  const single = Array.isArray(f) ? f[0] : f;
  return single && single.type === "field" && typeof single.value === "string"
    ? single.value
    : undefined;
}

export async function photosRoutes(app: FastifyInstance) {
  app.post("/footprints/:id/photos", { preHandler: requireUser }, async (req, reply) => {
    const { id } = req.params as { id: string };
    const footprint = await app.prisma.footprint.findUnique({ where: { id } });
    if (!footprint || footprint.userId !== req.userId) {
      return reply.code(404).send({ error: "not_found" });
    }
    const data = await req.file();
    if (!data) return reply.code(400).send({ error: "file_required" });
    const ext = EXT_BY_MIME[data.mimetype];
    if (!ext) return reply.code(415).send({ error: "unsupported_media_type" });

    const buffer = await data.toBuffer();
    const key = `photos/${footprint.id}/${randomUUID()}.${ext}`;
    await app.storage.put(key, buffer, data.mimetype);

    const takenAt = fieldValue(data.fields, "takenAt");
    const lat = fieldValue(data.fields, "lat");
    const lng = fieldValue(data.fields, "lng");
    const photo = await app.prisma.footprintPhoto.create({
      data: {
        footprintId: footprint.id,
        mediaUrl: key,
        takenAt: takenAt ? new Date(takenAt) : null,
        lat: lat ? Number(lat) : null,
        lng: lng ? Number(lng) : null,
      },
    });
    return reply.code(201).send({ id: photo.id, takenAt: photo.takenAt });
  });

  app.get("/photos/:id", { preHandler: requireUser }, async (req, reply) => {
    const { id } = req.params as { id: string };
    const photo = await app.prisma.footprintPhoto.findUnique({
      where: { id }, include: { footprint: { select: { userId: true } } },
    });
    if (!photo || photo.footprint.userId !== req.userId) {
      return reply.code(404).send({ error: "not_found" });
    }
    return { url: await app.storage.presignGet(photo.mediaUrl, 300) };
  });
}
```

```ts
async function queryPins(app: FastifyInstance) {
  const [places, counts] = await Promise.all([
    app.prisma.place.findMany({
      select: { id: true, name: true, level: true, parentId: true, lat: true, lng: true },
      orderBy: { createdAt: "asc" },
    }),
    app.prisma.card.groupBy({
      by: ["placeId"], where: { status: { not: "RETIRED" } }, _count: { _all: true },
    }),
  ]);
  const countBy = new Map(counts.map((c) => [c.placeId, c._count._all]));
  return places.map((p) => ({ ...p, cardCount: countBy.get(p.id) ?? 0 }));
}

// placesRoutes 内：
  app.get("/places/pins", async () => queryPins(app));

  app.get("/places/random", async (_req, reply) => {
    const withContent = (await queryPins(app)).filter((p) => p.cardCount > 0);
    if (withContent.length === 0) return reply.code(404).send({ error: "no_content" });
    return withContent[Math.floor(Math.random() * withContent.length)];
  });
```

```ts
import type { FastifyInstance } from "fastify";
import { requireAdmin } from "../auth.js";

export async function searchRoutes(app: FastifyInstance) {
  app.get("/search", async (req, reply) => {
    const { q } = req.query as { q?: string };
    const query = q?.trim() ?? "";
    if (!query) return reply.code(400).send({ error: "q_required" });

    const places = await app.prisma.place.findMany({
      where: { name: { contains: query, mode: "insensitive" } },
      select: { id: true, name: true, level: true, parentId: true, lat: true, lng: true },
      take: 20,
    });
    if (places.length === 0) {
      await app.prisma.searchMiss.upsert({
        where: { query },
        update: { count: { increment: 1 } },
        create: { query },
      });
      return [];
    }
    const counts = await app.prisma.card.groupBy({
      by: ["placeId"],
      where: { placeId: { in: places.map((p) => p.id) }, status: { not: "RETIRED" } },
      _count: { _all: true },
    });
    const countBy = new Map(counts.map((c) => [c.placeId, c._count._all]));
    return places.map((p) => ({ ...p, cardCount: countBy.get(p.id) ?? 0 }));
  });

  app.get("/admin/search-misses", { preHandler: requireAdmin }, async () => {
    return app.prisma.searchMiss.findMany({ orderBy: { count: "desc" }, take: 100 });
  });
}
```

```ts
import { databaseUrl } from "./env.js";
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { createPrisma, type PrismaClient } from "@zhixing/db";

type RegionsFile = {
  cities: Array<{
    name: string; lat: number; lng: number; intro?: string;
    spots: Array<{ name: string; lat: number; lng: number; intro?: string }>;
  }>;
};

export async function importRegions(
  prisma: PrismaClient,
  dataPath: string,
): Promise<{ cityCount: number; spotCount: number }> {
  const seed = JSON.parse(readFileSync(dataPath, "utf-8")) as RegionsFile;
  let spotCount = 0;
  for (const c of seed.cities) {
    let city = await prisma.place.findFirst({ where: { name: c.name, parentId: null } });
    if (!city) {
      city = await prisma.place.create({
        data: { name: c.name, level: "CITY", lat: c.lat, lng: c.lng, intro: c.intro },
      });
    }
    for (const s of c.spots) {
      await prisma.place.upsert({
        where: { name_parentId: { name: s.name, parentId: city.id } },
        update: { lat: s.lat, lng: s.lng, intro: s.intro },
        create: { ...s, level: "DISTRICT", parentId: city.id },
      });
      spotCount++;
    }
  }
  return { cityCount: seed.cities.length, spotCount };
}

const isMain = process.argv[1] === fileURLToPath(import.meta.url);
if (isMain) {
  const prisma = createPrisma(databaseUrl);
  const dataPath = fileURLToPath(new URL("../data/regions-places.json", import.meta.url));
  importRegions(prisma, dataPath)
    .then((r) => console.log(`imported: ${r.cityCount} cities + ${r.spotCount} spots`))
    .finally(() => prisma.$disconnect());
}
```

```ts
export async function generateCards(
  prisma: PrismaClient,
  llm: LlmClient,
  opts: { placeName: string; category: Category; count: number; webSearch: boolean },
): Promise<number> {
  const place = await prisma.place.findFirst({
    where: { name: opts.placeName },
    include: { parent: true },
  });
  if (!place) throw new Error(`地点不存在: ${opts.placeName}（先运行 import-places / import-regions）`);
  const cityName = place.parent?.name ?? place.name;

  let ok = 0;
  for (let i = 0; i < opts.count; i++) {
    const existing = await prisma.card.findMany({
      where: { placeId: place.id, status: { not: "RETIRED" } },
      select: { title: true },
    });
    const notes =
      opts.webSearch && llm.canWebSearch
        ? await llm.research(buildResearchPrompt(place.name, cityName, opts.category))
        : null;
    const draft = await writeCard(llm, {
      placeName: place.name,
      cityName,
      category: opts.category,
      notes,
      existingTitles: existing.map((c) => c.title),
    });
    const card = await saveDraft(prisma, place.id, opts.category, draft);
    console.log(`✓ [${CATEGORY_LABELS[opts.category]}] ${place.name}：${card.title} (${card.id})`);
    ok++;
  }
  return ok;
}
```

```ts
import { databaseUrl } from "./env.js";
import { parseArgs } from "node:util";
import { fileURLToPath } from "node:url";
import { createPrisma } from "@zhixing/db";
import { generateCards } from "./generate.js";
import { createLlm } from "./llm.js";
import { CATEGORY_LABELS, type Category } from "./prompts.js";

type PinLite = { id: string; name: string; cardCount: number };

export function pickJobs(
  pins: PinLite[],
  target: number,
  categories: Category[],
): Array<{ placeName: string; category: Category }> {
  const jobs: Array<{ placeName: string; category: Category }> = [];
  for (const pin of pins) {
    for (let n = pin.cardCount; n < target; n++) {
      jobs.push({ placeName: pin.name, category: categories[n % categories.length] });
    }
  }
  return jobs;
}

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

async function main() {
  // pnpm run 会把 "--" 透传进来，剥掉它（同 generate.ts 的处理）
  const rawArgs = process.argv.slice(2);
  const { values } = parseArgs({
    args: rawArgs[0] === "--" ? rawArgs.slice(1) : rawArgs,
    options: {
      fill: { type: "string" },
      city: { type: "string" },
      delay: { type: "string", default: "4000" },
      limit: { type: "string" },
      "no-web-search": { type: "boolean", default: false },
    },
  });
  const target = Number(values.fill);
  if (!Number.isInteger(target) || target < 1) {
    console.error("用法: pnpm --filter @zhixing/pipeline batch -- --fill <每地目标卡数> [--city 北京市] [--delay ms] [--limit N] [--no-web-search]");
    process.exit(1);
  }

  const prisma = createPrisma(databaseUrl);
  const llm = createLlm();
  try {
    // 城市本身与其下街区都参与补卡
    const where = values.city
      ? { OR: [{ parent: { name: values.city } }, { name: values.city }] }
      : {};
    const places = await prisma.place.findMany({ where, select: { id: true, name: true } });
    const counts = await prisma.card.groupBy({
      by: ["placeId"], where: { status: { not: "RETIRED" } }, _count: { _all: true },
    });
    const countBy = new Map(counts.map((c) => [c.placeId, c._count._all]));
    const pins: PinLite[] = places.map((p) => ({ id: p.id, name: p.name, cardCount: countBy.get(p.id) ?? 0 }));

    let jobs = pickJobs(pins, target, Object.keys(CATEGORY_LABELS) as Category[]);
    if (values.limit) jobs = jobs.slice(0, Number(values.limit));
    console.log(`计划生成 ${jobs.length} 张卡（target=${target}${values.city ? `，city=${values.city}` : ""}）`);

    let ok = 0, fail = 0;
    for (const job of jobs) {
      try {
        ok += await generateCards(prisma, llm, {
          placeName: job.placeName, category: job.category, count: 1,
          webSearch: !values["no-web-search"],
        });
      } catch (e) {
        fail++;
        console.error(`✗ ${job.placeName}/${job.category}: ${e instanceof Error ? e.message : e}`);
      }
      await sleep(Number(values.delay));
    }
    console.log(`完成：成功 ${ok}，失败 ${fail}`);
  } finally {
    await prisma.$disconnect();
  }
}

const isMain = process.argv[1] === fileURLToPath(import.meta.url);
if (isMain) {
  main().catch((e) => { console.error(e); process.exit(1); });
}
```
