```swift
import Foundation

enum APIError: Error, LocalizedError {
    case status(Int, String)
    var errorDescription: String? {
        if case let .status(code, body) = self { return "HTTP \(code)：\(body.prefix(120))" }
        return nil
    }
}

/// 与服务端 @fastify/multipart 兼容的 multipart/form-data 编码（纯函数，可单测）
func multipartBody(
    boundary: String, fields: [String: String],
    filename: String, mime: String, fileData: Data
) -> Data {
    var body = Data()
    func append(_ s: String) { body.append(Data(s.utf8)) }
    for (key, value) in fields.sorted(by: { $0.key < $1.key }) {
        append("--\(boundary)\r\ncontent-disposition: form-data; name=\"\(key)\"\r\n\r\n\(value)\r\n")
    }
    append("--\(boundary)\r\ncontent-disposition: form-data; name=\"file\"; filename=\"\(filename)\"\r\ncontent-type: \(mime)\r\n\r\n")
    body.append(fileData)
    append("\r\n--\(boundary)--\r\n")
    return body
}

@MainActor
final class APIClient {
    static let shared = APIClient()
    var token: String?

    static let decoder: JSONDecoder = {
        let withFraction = ISO8601DateFormatter()
        withFraction.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        let plain = ISO8601DateFormatter()
        let d = JSONDecoder()
        d.dateDecodingStrategy = .custom { decoder in
            let s = try decoder.singleValueContainer().decode(String.self)
            if let date = withFraction.date(from: s) ?? plain.date(from: s) { return date }
            throw DecodingError.dataCorrupted(.init(
                codingPath: decoder.codingPath, debugDescription: "无法解析日期 \(s)"))
        }
        return d
    }()

    static let encoder: JSONEncoder = {
        let e = JSONEncoder()
        e.dateEncodingStrategy = .iso8601
        return e
    }()

    func get<T: Decodable>(_ path: String) async throws -> T {
        try await send(path: path, method: "GET", body: nil, contentType: nil)
    }

    func post<T: Decodable>(_ path: String, json: some Encodable) async throws -> T {
        try await send(path: path, method: "POST",
                       body: try Self.encoder.encode(json), contentType: "application/json")
    }

    func delete(_ path: String) async throws {
        _ = try await raw(path: path, method: "DELETE", body: nil, contentType: nil)
    }

    func upload<T: Decodable>(
        _ path: String, file: Data, filename: String, mime: String, fields: [String: String]
    ) async throws -> T {
        let boundary = "zx-\(UUID().uuidString)"
        let body = multipartBody(boundary: boundary, fields: fields,
                                 filename: filename, mime: mime, fileData: file)
        return try await send(path: path, method: "POST", body: body,
                              contentType: "multipart/form-data; boundary=\(boundary)")
    }

    private func send<T: Decodable>(
        path: String, method: String, body: Data?, contentType: String?
    ) async throws -> T {
        let data = try await raw(path: path, method: method, body: body, contentType: contentType)
        return try Self.decoder.decode(T.self, from: data)
    }

    private func raw(
        path: String, method: String, body: Data?, contentType: String?
    ) async throws -> Data {
        guard let url = URL(string: path, relativeTo: AppConfig.apiBase) else {
            throw APIError.status(0, "非法路径 \(path)")
        }
        var req = URLRequest(url: url)
        req.httpMethod = method
        req.httpBody = body
        if let contentType { req.setValue(contentType, forHTTPHeaderField: "content-type") }
        if let token { req.setValue("Bearer \(token)", forHTTPHeaderField: "authorization") }
        let (data, response) = try await URLSession.shared.data(for: req)
        let code = (response as? HTTPURLResponse)?.statusCode ?? 0
        guard (200 ..< 300).contains(code) else {
            throw APIError.status(code, String(decoding: data, as: UTF8.self))
        }
        return data
    }
}
```

```swift
import Foundation
import Security

enum Keychain {
    private static func query(_ key: String) -> [String: Any] {
        [kSecClass as String: kSecClassGenericPassword,
         kSecAttrService as String: "app.zhixing.session",
         kSecAttrAccount as String: key]
    }

    static func save(_ value: String, key: String) {
        SecItemDelete(query(key) as CFDictionary)
        var attrs = query(key)
        attrs[kSecValueData as String] = Data(value.utf8)
        SecItemAdd(attrs as CFDictionary, nil)
    }

    static func load(key: String) -> String? {
        var q = query(key)
        q[kSecReturnData as String] = true
        q[kSecMatchLimit as String] = kSecMatchLimitOne
        var out: AnyObject?
        guard SecItemCopyMatching(q as CFDictionary, &out) == errSecSuccess,
              let data = out as? Data else { return nil }
        return String(decoding: data, as: UTF8.self)
    }

    static func delete(key: String) {
        SecItemDelete(query(key) as CFDictionary)
    }
}
```

`apps/ios/Zhixing/Core/SessionStore.swift`:
```swift
import Foundation
import Observation

@MainActor
@Observable
final class SessionStore {
    private(set) var user: User?
    private(set) var token: String?
    private(set) var restoring = true

    var isSignedIn: Bool { token != nil }

    func restore() async {
        defer { restoring = false }
        guard let saved = Keychain.load(key: "token") else { return }
        token = saved
        APIClient.shared.token = saved
        do {
            user = try await APIClient.shared.get("/me")
        } catch let APIError.status(code, _) where code == 401 {
            signOut() // token 过期才登出；网络错误保持登录态
        } catch {}
    }

    private struct AppleBody: Encodable { let identityToken: String; let nickname: String? }
    func signInWithApple(identityToken: String, nickname: String?) async throws {
        apply(try await APIClient.shared.post(
            "/auth/apple", json: AppleBody(identityToken: identityToken, nickname: nickname)))
    }

    private struct DevBody: Encodable { let appleSub: String }
    func devLogin(sub: String) async throws {
        apply(try await APIClient.shared.post("/auth/dev", json: DevBody(appleSub: sub)))
    }

    func signOut() {
        token = nil
        user = nil
        APIClient.shared.token = nil
        Keychain.delete(key: "token")
    }

    private func apply(_ auth: AuthResponse) {
        token = auth.token
        user = auth.user
        APIClient.shared.token = auth.token
        Keychain.save(auth.token, key: "token")
    }
}
```

```swift
import SwiftUI

final class PlaceAnnotation: NSObject, MKAnnotation {
    let pin: Pin
    let isFootprint: Bool
    init(pin: Pin, isFootprint: Bool) {
        self.pin = pin
        self.isFootprint = isFootprint
    }
    var coordinate: CLLocationCoordinate2D {
        CLLocationCoordinate2D(latitude: pin.lat, longitude: pin.lng)
    }
    var title: String? { pin.name }
}

struct MapView: UIViewRepresentable {
    let pins: [Pin]
    let litPlaceIds: Set<String>
    let showHumanities: Bool
    let showFootprints: Bool
    @Binding var flyTo: Pin?
    var onSelect: (Pin) -> Void

    func makeUIView(context: Context) -> MKMapView {
        let map = MKMapView()
        map.delegate = context.coordinator
        map.register(MKMarkerAnnotationView.self,
                     forAnnotationViewWithReuseIdentifier: MKMapViewDefaultAnnotationViewReuseIdentifier)
        map.register(MKMarkerAnnotationView.self,
                     forAnnotationViewWithReuseIdentifier: MKMapViewDefaultClusterAnnotationViewReuseIdentifier)
        map.pointOfInterestFilter = .excludingAll
        map.setRegion(MKCoordinateRegion(
            center: CLLocationCoordinate2D(latitude: 36.0, longitude: 105.0),
            span: MKCoordinateSpan(latitudeDelta: 32, longitudeDelta: 32)), animated: false)
        return map
    }

    func updateUIView(_ map: MKMapView, context: Context) {
        context.coordinator.parent = self
        context.coordinator.sync(map: map)
        if let target = flyTo {
            map.setRegion(MKCoordinateRegion(
                center: CLLocationCoordinate2D(latitude: target.lat, longitude: target.lng),
                span: MKCoordinateSpan(latitudeDelta: 0.08, longitudeDelta: 0.08)), animated: true)
            DispatchQueue.main.async { flyTo = nil }
        }
    }

    func makeCoordinator() -> Coordinator { Coordinator(self) }

    final class Coordinator: NSObject, MKMapViewDelegate {
        var parent: MapView
        private var syncedKey = ""
        init(_ parent: MapView) { self.parent = parent }

        /// 数据没变就不重建标注（updateUIView 会被 SwiftUI 频繁调用）
        func sync(map: MKMapView) {
            let key = "\(parent.pins.count)-\(parent.litPlaceIds.count)-\(parent.showHumanities)-\(parent.showFootprints)"
            guard key != syncedKey else { return }
            syncedKey = key
            map.removeAnnotations(map.annotations)
            var annotations: [PlaceAnnotation] = []
            if parent.showHumanities {
                annotations += parent.pins.filter { $0.cardCount > 0 }
                    .map { PlaceAnnotation(pin: $0, isFootprint: false) }
            }
            if parent.showFootprints {
                annotations += parent.pins.filter { parent.litPlaceIds.contains($0.id) }
                    .map { PlaceAnnotation(pin: $0, isFootprint: true) }
            }
            map.addAnnotations(annotations)
        }

        func mapView(_ mapView: MKMapView, viewFor annotation: MKAnnotation) -> MKAnnotationView? {
            if annotation is MKClusterAnnotation {
                let v = mapView.dequeueReusableAnnotationView(
                    withIdentifier: MKMapViewDefaultClusterAnnotationViewReuseIdentifier,
                    for: annotation) as! MKMarkerAnnotationView
                v.markerTintColor = .systemTeal
                return v
            }
            guard let pa = annotation as? PlaceAnnotation else { return nil }
            let v = mapView.dequeueReusableAnnotationView(
                withIdentifier: MKMapViewDefaultAnnotationViewReuseIdentifier,
                for: annotation) as! MKMarkerAnnotationView
            v.clusteringIdentifier = pa.isFootprint ? "footprint" : "place"
            v.markerTintColor = pa.isFootprint ? .systemOrange : .systemTeal
            v.glyphText = pa.isFootprint ? "行" : "知"
            v.displayPriority = .defaultHigh
            return v
        }

        func mapView(_ mapView: MKMapView, didSelect view: MKAnnotationView) {
            if let cluster = view.annotation as? MKClusterAnnotation {
                mapView.showAnnotations(cluster.memberAnnotations, animated: true)
                mapView.deselectAnnotation(cluster, animated: false)
                return
            }
            if let pa = view.annotation as? PlaceAnnotation {
                mapView.deselectAnnotation(pa, animated: false) // 允许再次点击
                parent.onSelect(pa.pin)
            }
        }
    }
}
```

```swift
import Observation
import UIKit

@MainActor
@Observable
final class FootprintStore {
    private(set) var footprints: [Footprint] = []
    private(set) var stats: Stats?

    var litPlaceIds: Set<String> { Set(footprints.map(\.placeId)) }

    func load() async {
        if let list: [Footprint] = try? await APIClient.shared.get("/me/footprints") {
            footprints = list
        }
        stats = try? await APIClient.shared.get("/me/stats")
    }

    private struct CreateBody: Encodable {
        let placeId: String
        let text: String?
        let date: Date
    }

    func create(placeId: String, text: String?, date: Date) async throws -> Footprint {
        let fp: Footprint = try await APIClient.shared.post(
            "/footprints", json: CreateBody(placeId: placeId, text: text, date: date))
        footprints.insert(fp, at: 0)
        return fp
    }

    /// 统一转 JPEG（PhotosPicker/相册可能给 HEIC；顺带压到上传友好的体积）
    func uploadPhoto(
        footprintId: String, imageData: Data,
        takenAt: Date? = nil, lat: Double? = nil, lng: Double? = nil
    ) async throws {
        let jpeg = UIImage(data: imageData)?.jpegData(compressionQuality: 0.85) ?? imageData
        var fields: [String: String] = [:]
        if let takenAt { fields["takenAt"] = ISO8601DateFormatter().string(from: takenAt) }
        if let lat { fields["lat"] = String(lat) }
        if let lng { fields["lng"] = String(lng) }
        let _: FootprintPhotoMeta = try await APIClient.shared.upload(
            "/footprints/\(footprintId)/photos",
            file: jpeg, filename: "photo.jpg", mime: "image/jpeg", fields: fields)
    }

    func delete(_ footprint: Footprint) async {
        try? await APIClient.shared.delete("/footprints/\(footprint.id)")
        footprints.removeAll { $0.id == footprint.id }
        stats = try? await APIClient.shared.get("/me/stats")
    }
```

```swift
import Foundation
import Observation
import Photos
import UIKit

struct AssetMeta: Identifiable, Equatable {
    let id: String // PHAsset.localIdentifier
    let takenAt: Date
    let lat: Double
    let lng: Double
}

struct DraftGroup: Identifiable {
    let id = UUID()
    let pin: Pin
    var assets: [AssetMeta] // 按 takenAt 升序
    var include = true
    var start: Date { assets.first?.takenAt ?? .now }
}

func haversineMeters(lat1: Double, lng1: Double, lat2: Double, lng2: Double) -> Double {
    let r = 6_371_000.0
    let dLat = (lat2 - lat1) * .pi / 180
    let dLng = (lng2 - lng1) * .pi / 180
    let a = sin(dLat / 2) * sin(dLat / 2)
        + cos(lat1 * .pi / 180) * cos(lat2 * .pi / 180) * sin(dLng / 2) * sin(dLng / 2)
    return 2 * r * atan2(sqrt(a), sqrt(1 - a))
}

/// 匹配半径按层级（规格书 §4.3 自建地点库半径匹配）
func matchRadius(level: String) -> Double {
    switch level {
    case "DISTRICT", "POI": return 3_000
    case "CITY": return 30_000
    default: return 0 // PROVINCE 不直接挂照片
    }
}

/// 距离/半径比最小者胜出——街区半径小，同时命中时自然优先于城市
func nearestPin(lat: Double, lng: Double, pins: [Pin]) -> Pin? {
    var best: (pin: Pin, score: Double)?
    for pin in pins {
        let radius = matchRadius(level: pin.level)
        guard radius > 0 else { continue }
        let d = haversineMeters(lat1: lat, lng1: lng, lat2: pin.lat, lng2: pin.lng)
        guard d <= radius else { continue }
        let score = d / radius
        if best == nil || score < best!.score { best = (pin, score) }
    }
    return best?.pin
}

/// 按地点聚类；同地点内相邻照片间隔 > gap（默认 48h）拆成两段旅程
func clusterAssets(_ assets: [AssetMeta], pins: [Pin], gap: TimeInterval = 48 * 3600) -> [DraftGroup] {
    var byPin: [String: (pin: Pin, assets: [AssetMeta])] = [:]
    for asset in assets {
        guard let pin = nearestPin(lat: asset.lat, lng: asset.lng, pins: pins) else { continue }
        byPin[pin.id, default: (pin, [])].assets.append(asset)
    }
    var groups: [DraftGroup] = []
    for entry in byPin.values {
        let sorted = entry.assets.sorted { $0.takenAt < $1.takenAt }
        var current: [AssetMeta] = []
        for asset in sorted {
            if let last = current.last, asset.takenAt.timeIntervalSince(last.takenAt) > gap {
                groups.append(DraftGroup(pin: entry.pin, assets: current))
                current = []
            }
            current.append(asset)
        }
        if !current.isEmpty {
            groups.append(DraftGroup(pin: entry.pin, assets: current))
        }
    }
    return groups.sorted { $0.start > $1.start }
}

@MainActor
@Observable
final class AlbumImportModel {
    enum Phase {
        case scanning
        case denied
        case review
        case importing(done: Int, total: Int)
        case finished(created: Int, photos: Int)
    }

    var phase: Phase = .scanning
    var groups: [DraftGroup] = []
    var unmatchedCount = 0

    func scan(pins: [Pin]) async {
        let status = await PHPhotoLibrary.requestAuthorization(for: .readWrite)
        guard status == .authorized || status == .limited else {
            phase = .denied
            return
        }
        let assets = Self.fetchAssetMetas()
        let matched = clusterAssets(assets, pins: pins)
        unmatchedCount = assets.count - matched.reduce(0) { $0 + $1.assets.count }
        groups = matched
        phase = .review
    }

    nonisolated static func fetchAssetMetas() -> [AssetMeta] {
        let options = PHFetchOptions()
        options.predicate = NSPredicate(format: "mediaType == %d", PHAssetMediaType.image.rawValue)
        let result = PHAsset.fetchAssets(with: options)
        var metas: [AssetMeta] = []
        result.enumerateObjects { asset, _, _ in
            guard let location = asset.location, let date = asset.creationDate else { return }
            metas.append(AssetMeta(
                id: asset.localIdentifier, takenAt: date,
                lat: location.coordinate.latitude, lng: location.coordinate.longitude))
        }
        return metas
    }

    func runImport(store: FootprintStore) async {
        let chosen = groups.filter(\.include)
        var created = 0
        var photosUploaded = 0
        for (index, group) in chosen.enumerated() {
            phase = .importing(done: index, total: chosen.count)
            do {
                let fp = try await store.create(
                    placeId: group.pin.id, text: nil, date: group.start)
                for asset in group.assets.prefix(9) { // 每段足迹照片上限 9 张
                    if let data = await Self.imageData(localIdentifier: asset.id) {
                        try await store.uploadPhoto(
                            footprintId: fp.id, imageData: data,
                            takenAt: asset.takenAt, lat: asset.lat, lng: asset.lng)
                        photosUploaded += 1
                    }
                }
                created += 1
            } catch {
                continue // 单组失败不阻塞整体
            }
        }
        await store.load()
        phase = .finished(created: created, photos: photosUploaded)
    }

    nonisolated static func imageData(localIdentifier: String) async -> Data? {
        guard let asset = PHAsset.fetchAssets(
            withLocalIdentifiers: [localIdentifier], options: nil).firstObject else { return nil }
        return await withCheckedContinuation { continuation in
            let options = PHImageRequestOptions()
            options.deliveryMode = .highQualityFormat // 单次回调，防 continuation 重复 resume
            options.isNetworkAccessAllowed = true // iCloud 照片可下载
            PHImageManager.default().requestImage(
                for: asset, targetSize: CGSize(width: 1600, height: 1600),
                contentMode: .aspectFit, options: options
            ) { image, _ in
                continuation.resume(returning: image?.jpegData(compressionQuality: 0.8))
            }
        }
    }
}
```
